function openGroup(panelId){
  tabs.forEach(t=>t.classList.toggle('active',t.dataset.panel===panelId));
  document.getElementById('en-panel').classList.toggle('active',panelId==='en-panel');
  document.getElementById('jp-panel').classList.toggle('active',panelId==='jp-panel');

  const target=navs.find(n=>n.dataset.group===group);
  if(target) openArticle(target.dataset.target,false);
  localStorage.setItem('daily-panel',panelId);
}
function openArticle(id,scroll=true){
  articles.forEach(a=>a.classList.toggle('active',a.id===id));
  navs.forEach(n=>n.classList.toggle('active',n.dataset.target===id));
  const group=document.getElementById(id)?.id.startsWith('en')?'en-panel':'jp-panel';
  document.getElementById('en-panel').classList.toggle('active',group==='en-panel');
  document.getElementById('jp-panel').classList.toggle('active',group==='jp-panel');
  document.getElementById('en-overview').classList.toggle('active',group==='en-panel');
  document.getElementById('jp-overview').classList.toggle('active',group==='jp-panel');
  tabs.forEach(t=>t.classList.toggle('active',t.dataset.panel===group));
  localStorage.setItem('daily-article',id);
  if(scroll) window.scrollTo({top:0,behavior:'smooth'});
}
tabs.forEach(t=>t.addEventListener('click',()=>openGroup(t.dataset.panel)));
navs.forEach(n=>n.addEventListener('click',()=>openArticle(n.dataset.target)));

document.querySelectorAll('.lang-btn').forEach(btn=>btn.addEventListener('click',()=>{

  const lang=btn.dataset.lang;
  article.querySelectorAll('.lang').forEach(box=>{
    const show=lang==='all'||box.classList.contains(lang);
    box.classList.toggle('hidden',!show);
  });
}));

document.querySelectorAll('.answer-btn').forEach(btn=>btn.addEventListener('click',()=>{
